// ARRAY OPERATIONS
// if a field holds an array, we can query for:
// an exact match on the array
// or specific values in the array
// if the array itself holds embedded documents, we can query for specific
// fields in the embedded documents using dot notation

// using this format for documents in a collection
    {"_id": 5, "type": "food", "item": "cake",   "ratings": [5, 8, 9]}

// exact array match. uses the equals to operator
db.inventory.find({
    ratings: [5, 8, 9]
})

//matching an array element, dont use []
// returns all documents that have a '5' as at least one element
db.inventory.find({
    ratings: 5
})

//can specify multiple criteria
//this works a bit weird, one element can satisfy one condition
//while another satisfies the other condition
//or you can have a single element satisfying both
// [1, 10] will return
// [6] will also return 
db.inventory.find({
    ratings: {
        $gt: 5,
        $lt: 9
    }
})

// this can be made less weird by using an $elemMatch
// will only return where the ratings array contains at least
// one element greater than 5 AND less than 9
// [1, 10] nope
// [6] yep
db.inventory.find({
    ratings: {
        $elemMatch: {
            $gt 5,
            $lt 9
        }
    }
})

//searching for text: if we have a longer string as a field, and we want to basically grep item

//text indexes are needed when using $text to search the contents of strings
//these indexes can be applied to any string field, or array of string elements

//apply a text index to the comments field
db.createIndex({comments: "text"})

// use the dropIndex() method to clear the index

//search for documents that have the word "big" in the comments. case insensitive by default
var c = db.blogs.find({
    $text: {$search: "big"}
})
while (c.hasNext()){
    printjson(c.next())
}

//can search for multiple strings, works like an OR
//documents with the word 'big' or 'data'
var c = db.blogs.find({
    $text: {$search: "big data"}
})
while (c.hasNext()){
    printjson(c.next())
}

//can use wildcard specifier ($**) to index all fields
//in the collection with string content
//useful for really messy unstructured data but it can negatively affect performance 
db.collection.createIndex({ "S**": "text" })

//text search within the aggregation pipeline
//can be done using:
//the $text query operator when in the $match stage

//the $match including a $text must be the FIRST stage in the pipeline
//a text operator can only occur once in this stage
//and the text operator cannot appear in an $or or $not expression

//seach for the term 'nosql' in the $match stage
//then calculate the total likes for the matching documents in the $group stage
db.blogs.aggregate([
    {
        $match: {
            $text: {$search: "nosql}
        }
    }
    {
        $group: {
            _id: null,
            totalLikes: { $sum: "$likes"}
        }
    }
])

function insertData(dbName, colName, num) {
    const col = db.getSiblingDB(dbName).getCollection(colName);
    for (i = 0; i < num; i++){
        col.insertOne([x: i]);
    }
    print(col.countDocuments())
}

//recall a query to compute the total number of likes
//this is a pain in the ass having a raw query whenever this aggregate is needed
//generalise it into a function
function computeTotalLikes(dbName, colName, blogTitle){
    var col = db.getSiblingDB(dbName).getCollection(colName);

    var res = col.aggregate([
        {
            $match: {title: blogTitle}
        },
        {
            $group: {
                _id: blotTitle,
                num_likes: {$sum: "$likes"}
            }
        }
    ]);

    //print out all the results
    while (res.hasNext()){
        printjson(res.next());
    }
}
