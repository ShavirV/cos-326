// ARRAY OPERATIONS
// if a field holds an array, we can query for:
// an exact match on the array
// or specific values in the array
// if the array itself holds embedded documents, we can query for specific
// fields in the embedded documents using dot notation

// using this format for documents in a collection
    {"_id": 5, "type": "food", "item": "cake",   "ratings": [5, 8, 9]}

// exact array match. uses the equals to operator
db.inventory.find({
    ratings: [5, 8, 9]
})

//matching an array element, dont use []
// returns all documents that have a '5' as at least one element
db.inventory.find({
    ratings: 5
})

//can specify multiple criteria
//this works a bit weird, one element can satisfy one condition
//while another satisfies the other condition
//or you can have a single element satisfying both
// [1, 10] will return
// [6] will also return 
db.inventory.find({
    ratings: {
        $gt: 5,
        $lt: 9
    }
})

// this can be made less weird by using an $elemMatch
// will only return where the ratings array contains at least
// one element greater than 5 AND less than 9
// [1, 10] nope
// [6] yep
db.inventory.find({
    ratings: {
        $elemMatch: {
            $gt 5,
            $lt 9
        }
    }
})

//an array of embedded documents
{
    _id: 100,
    type: "food",
    item: "xyz",
    qty : 25, 
    price: 2.5,

    ratings: [ 5, 8, 9 ],

    // an array of embedded documents
    memos: [
        { 
            memo: "on time",
            by: "shipping" 
        },
        { 
            memo: "approved",
            by: "billing" 
        }
    ]       
}

//select all documents where the memos field contains an array
//whose first element (memos.0) is a document with the field 'by',
//which has the value 'shipping'
db.inventory.find({
    'memos.0.by': 'shipping'
})

//AGGREGATION 

db.<collection>.aggregate(OPERATION)

//These functions are defined:
$sum
$avg
$min
$max

// there is also an aggregation pipeline that can be applied
// at each step a new operation is done before passing to the next step

db.<collection>.aggregate(OPERATION)

// apply an array of aggregation stages
db.collection.aggregate([
    {$match: <doc criterion>},
    {$match: <group spec>},
    {$match: <sort spec>}
])

// compute the total number of 'likes' for each title 
// assuming an array {likes: [1, 2, 3]}
db.blogs.aggregate([
    {
        $group:
        {
            _id: "$title,
            num_likes: {$sum: "$likes}
        }
    }
])

//adding on, computing the number of likes
//for the title neo4j
db.blogs.aggregate({
    {$match: {title: "neo4j"}}
    {$group:
        {
            _id: "$title",
            num_likes: {$sum: "$likes"}
        }    
    }
})

//same thing but sort in ascending order of number of likes
db.blogs.aggregate([
    {
        $group:
        {
            _id: "$title,
            num_likes: {$sum: "$likes}
        }
    }
    {$sort: {"num_likes": 1}}
])
