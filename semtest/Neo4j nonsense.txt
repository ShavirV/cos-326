// we store data using simple graph theory concepts:
// Nodes - data objects
// Relationships - edges connecting Nodes
// Properties - named data values (attributes are expressed as key value pairs)

// represent social networks, generate reccomendations

// uses Cypher as its query language

//labels can be used to associate a group of Nodes
//can group nodes together by applying the same label to each member
//for instance all people can be given the label 'Person'

//A Neo4j relationship:

- start node
- end node
- exactly one relationship type
- an optional set of Properties
- a direction

For instance, a FOLLOWS relationship can have a property 'since', associated with a year

Cypher is declarative like SQL, describing what to do, not how to do it.

//CREATION

CREATE(psT:Person {
    name: "shav",
    from: "pta",
    hobby: "notthis"
})

//() used to indicate a node
// psT:Person, a var psT, label 'Person' for the new node
// {} properties for the node


//CREATE does not check for whether data already exists or not, it just adds the thing as a new node
//use a MERGE when we dont want to create duplicate nodes

CREATE 
    (psJ:Person { name: "johan", from: "pta", hobby: "surfing"}),
    (psN:Person { name: "neo", from: "cpt", hobby: "soccer"}),

    (psT)-[:FOLLOWS {since: 2012}]->(psJ),
    (psT)-[:FOLLOWS {since: 2009}]->(psN),
    (psJ)-[:FOLLOWS {since: 2010}]->(psT)

//use (from)-[:rel {relattr}]->(to)
//to define a relationship between nodes
// its like a stupid lil picture of a graph with an arrow and relationship
// ()-[]->()

//MATCH - finding nodes

MATCH(psT:Person)           //specify a pattern of nodes an relationships
WHERE psT.name = "shav",    //constrain the results 
RETURN psT;                 //what do we return 

// can use MATCH to get a whole chunk of a graph at once?

MATCH (n)-[r]->(m)
RETURN n, r, m

// this kinda works like prolog in a weird way
// pattern matching
// we can describe what to find in the graph
// for instance the people that shav FOLLOWS

// genuinely just predicates and that nonsense

MATCH (psN:Person)->[:FOLLOWS]->(followed)
WHERE psN.name = "shav"
RETURN psN, followed

// (followed) will be bound to the people that psN has the relation FOLLOWS with
// we then provide a restriction for which nodes can actually be psN

// i think it can also work in the opposite way

MATCH (follows)->[:FOLLOWS]->(psN)
WHERE psN.name = "shav"
RETURN psN, follows

//now this should return the people that follow shav

//CRUD clauses:

- CREATE
- MERGE
- SET       - update a node or rel
- DELETE    - delete the node or rel
- REMOVE    - remove properties from elements
- FOREACH   - iterate over a path, or a collection created by aggregation
- DETACH / DELETE - delete a node or a set of nodes. also deletes associated relationships

// SET using MATCH to find a specific node
MATCH (psN:Person)-[r:FOLLOWS]->(psT:Person)
WHERE psN.name = "neo" AND psT.name = "Thandi" // so we know theres only really one relation
SET r.since = 2018 // update the property of the relation between these nodes
RETURN psN, r, psT

// MERGE can be used only ON CREATE or ON MATCH or both

MERGE (psT:Person [name: "Thandi"])
ON MATCH SET psT.age = 21 
ON CREATE SET 
    psT.from = "durban",
    psT.hobby = "singing",
    psT.age = 21
RETURN psT; 

// ON MATCH SET means if we see this node in existence, set the age field to 21
// ON CREATE SET means if this node does not exist, set all of those fields to those values


// DELETE

// Deleting a relationship
// We use ()-[]->() to find the relationship between these two nodes
// If any such relation exists, we store it in r and then delete it
MATCH (psT:Person {name: "Thandi})-[r:FOLLOWS]->(psN:Person {name: "neo"})
DELETE r;

// Deleting a node and its connected relationships
// DELETE on its own cannot delete a node that still has relationships
// DETACH DELETE removes this node and its connected relationships
// almost like a weird version of memory management in c++

MATCH (p:Person {name: "zorro"})
DETACH DELETE p;

// DELETE for graph elements (nodes and relationships)
// delete all nodes of group Person and all connected relationships
MATCH (psN:Person)
DETACH DELETE psn;

//REMOVE

//sort of like a lighter delete
//remove properties and labels from graph elements
//also an alternate way of matching by a certain field
//after this zorro will have no hobby field
MATCH (psZ:Person)
WHERE psz.name = "zorro"
REMOVE psZ.hobby
return psZ;

//READING clauses

MATCH - Allows you to specify patterns that Cypher will search for in the DB
OPTIONAL MATCH - If no matches are found, will use NULLS for missing parts of the pattern
WHERE - Filter the results of MATCH, OPTIONAL MATCH, & with
AGGREGATION clauses - works like group by in SQL

MATCH (<var>) //resolves to any node
MATCH (<var> : <label>) //single node pattern, matches will be all nodes of type <label>
MATCH (<var1> {key:val})-->(<var2>) //specifies a path from var 1 to var 2 -->

MATCH (psT:Person) RETURN psT; //entire set of nodes
MATCH (psT:Person) RETURN psT.name //names for each node in label

// ORDER BY ussed for soring in a MATCH

MATCH(psn) RETURN psn.name ORDER BY psn.name;
