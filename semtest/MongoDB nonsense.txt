MongoDB nonsense:

use mydb // create or open
mydb.createCollection(newCollection) // create a new collection
db.myCollection.insertOne({name: "value", x: 3})
db.myCollection.find({x:3})

// basic CRUD operations
// CREATION

// BSON object
doc = { name: "value", x: 3}

//creating a collection
db.createCollection("myCollection")
db.myCollection.insertOne(doc)

// we can also implicitly create a collection by inserting without initialisation
db.myCollection.insertOne({name: "value", x: 3})

// delete a collection
db.myCollection.drop()

// INSERTION

done by either insertOne() or insertMany() into a collection
insertMany() works using array notation (array of BSON objects)

db.myCollection.insertOne({
	name: "mongo",
	x: 3
})

// use [{array}, {notation}]
db.myCollection.insertMany([
	{ name: "mongo", x: 3 },
	{ name: "neo4j", x: 4 },
	... more objects
])


// can also specify the _id field, else mongo will generate one for you
db.myCollection.insertOne({
	_id : objectId("somebullshitstring"),
	name: "mongo",
	x: 3
})

// QUERYING - Relational operators

db.myCollection.find({ x:3 }) //colon used for equality

db>myCollection.find({ x : {$lt: 3}})
// less than - { key : { $lt: value } }
// less than or equal to { key : { $lte: value } }
// basically the same thing
// $gt, $gte, $ne //not equal


// logical operators

// find all documents where name is mongo AND n is 3
db.myCollection.find({
$and: [
	{ name: "mongo" },
	{ n: 3}
]
})

//similar for OR
db.myCollection.find({
$or: [
	{ name: "mongo" },
	{ name: "neo4j"},
]
})

// alternative method for or
// think of it almost like a foreach syntactically, looks for matches based on the array after $ in : [...]
db.myCollection.find(
[ name:
	[ $in :
		["mongo", "neo4j"]
	]
]
)


//use the it helper to get more results for find()

// returns all items in the collection
db.myCollection.find() 

db.myCollection.find( { <criterion> },
	{
		<field1> : <selector>,
		name: 1, // selector == 1 therefore include name
		_id: 0, // selector == 0 therefore exclude id
	}
)

// NOTE: you CANNOT use exclusion and inclusion in the same projection, except for _id
// this means you either declare what to include or what to exclude, the interpreter
// will infer what to do with fields that are not mentioned otherwise
db.myCollection.find(
	{x : { $gt: 2 } },

	{
		name: 1,
		_id: 0,
	}
)


// SORTING FIND
<find query>.sort(
	<field1>:<order>,
	<field2>: 1 // 1 for ascending order, 2 for descending
)


// find items where x field is > 2, and sort by name ascending
db.myCollection.find(
	{
		x: {$gt: 2}
	}
).sort(
	{
		name: 1
	}
)

// UPDATING documents
db.myCollection.updateOne(
	{ <field> : { <criterion> }},
	{ $set: { <action> }}
	{ multi: <selector>} // if true, then more than one document can be updated
)

// find one document (the first matching document) where x == 3,
// and update its name field to be "mongodb"
db.myCollection.updateOne(
	{ x: 3 },
	{ $set : {name: "mongodb"}}
)

// same as the previous query except we change ALL matching documents
db.myCollection.updateMany(
	{ x : 3},
	{ $set: {name: "mongodb" }}
)

// DELETING documents
// pretty simple its the exact same as updating but without the $set since we obviously take the document out back and chuck it

db.myCollection.deleteMany(
	{ <field> : <criterion> }
)

db.myCollection.deleteOne(
	{ x : 3}
)

// QUERYING find: indexing 
// indexing works the same as in other DBs
// speeds up find, update, and remove as well as SORTING
// but adds storage and write overhead (since new writes cant just be chucked, and there may be more redundancy)

//create an index on a field
// 1 == ascending
db.myCollection.createIndex( { name: 1})

//compound index
db.myCollection.createIndex( { name: 1, x: -1})

// can also analyse queries
// will examine the query plan and determine if an index is being used
db.myCollection.find({ name: "mongo"}).explain("executionStats")
